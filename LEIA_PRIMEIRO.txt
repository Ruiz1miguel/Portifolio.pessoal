-- Execute no SQL Editor do Supabase.
-- IMPORTANTE: substitua o UUID no INSERT final pelo User UID do seu administrador.
create table if not exists public.projetos (
    id bigint generated by default as identity primary key,
    titulo text not null check (char_length(titulo) between 1 and 160),
    descricao text not null,
    tecnologias text not null,
    link_git text not null default ''
);
create table if not exists public.mensagens (
    id bigint generated by default as identity primary key,
    nome text not null check (char_length(nome) between 1 and 120),
    email text not null check (char_length(email) between 3 and 254),
    texto text not null check (char_length(texto) between 1 and 3000),
    data_envio timestamptz not null default now()
);
create table if not exists public.admins (
    user_id uuid primary key references auth.users(id) on delete cascade
);
alter table public.projetos enable row level security;
alter table public.mensagens enable row level security;
alter table public.admins enable row level security;

-- Projetos: leitura pública, escrita somente para administradores autenticados.
create policy "Projetos visiveis publicamente" on public.projetos for select to anon, authenticated using (true);
create policy "Admin insere projetos" on public.projetos for insert to authenticated with check
(exists(select 1 from public.admins where user_id = (select auth.uid())));
create policy "Admin edita projetos" on public.projetos for update to authenticated using
(exists(select 1 from public.admins where user_id = (select auth.uid()))) with check
(exists(select 1 from public.admins where user_id = (select auth.uid())));
create policy "Admin exclui projetos" on public.projetos for delete to authenticated using
(exists(select 1 from public.admins where user_id = (select auth.uid())));

-- Qualquer visitante pode enviar mensagem, mas somente admins podem lê-la ou apagá-la.
create policy "Publico envia mensagens" on public.mensagens for insert to anon, authenticated with check (true);
create policy "Admin le mensagens" on public.mensagens for select to authenticated using
(exists(select 1 from public.admins where user_id = (select auth.uid())));
create policy "Admin exclui mensagens" on public.mensagens for delete to authenticated using
(exists(select 1 from public.admins where user_id = (select auth.uid())));
create policy "Admin visualiza permissao" on public.admins for select to authenticated using
(user_id = (select auth.uid()));
